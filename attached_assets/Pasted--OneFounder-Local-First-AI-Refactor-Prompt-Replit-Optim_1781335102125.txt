```typescript
// packages/ai/providers/ollama.ts

export interface OllamaConfig {
  baseUrl: string;       // e.g. http://localhost:11434 or ngrok URL
  model: string;         // e.g. qwen3:8b
  temperature?: number;
  contextWindow?: number;
}

export interface ChatMessage {
  role: 'system' | 'user' | 'assistant';
  content: string;
}

export interface StreamChunk {
  content: string;
  done: boolean;
  model?: string;
  totalTokens?: number;
}

export class OllamaProvider {
  private config: OllamaConfig;

  constructor(config: OllamaConfig) {
    this.config = config;
  }

  // Non-streaming completion
  async chat(messages: ChatMessage[]): Promise<string> {
    const response = await fetch(`${this.config.baseUrl}/api/chat`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        model: this.config.model,
        messages,
        stream: false,
        options: {
          temperature: this.config.temperature ?? 0.7,
          num_ctx: this.config.contextWindow ?? 4096,
        },
      }),
    });

    if (!response.ok) {
      const error = await response.text();
      throw new OllamaError(`Ollama request failed: ${response.status} – ${error}`);
    }

    const data = await response.json();
    return data.message?.content ?? '';
  }

  // Streaming completion — yields partial chunks
  async *chatStream(messages: ChatMessage[]): AsyncGenerator<StreamChunk> {
    const response = await fetch(`${this.config.baseUrl}/api/chat`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        model: this.config.model,
        messages,
        stream: true,
        options: {
          temperature: this.config.temperature ?? 0.7,
          num_ctx: this.config.contextWindow ?? 4096,
        },
      }),
    });

    if (!response.ok) {
      throw new OllamaError(`Stream failed: ${response.status}`);
    }

    const reader = response.body!.getReader();
    const decoder = new TextDecoder();

    while (true) {
      const { done, value } = await reader.read();
      if (done) break;

      const lines = decoder.decode(value, { stream: true }).split('\n').filter(Boolean);
      for (const line of lines) {
        try {
          const chunk = JSON.parse(line);
          yield {
            content: chunk.message?.content ?? '',
            done: chunk.done ?? false,
            model: chunk.model,
            totalTokens: chunk.prompt_eval_count,
          };
          if (chunk.done) return;
        } catch {
          // Malformed chunk — skip
        }
      }
    }
  }

  // Health check
  async ping(): Promise<{ ok: boolean; models: string[] }> {
    try {
      const res = await fetch(`${this.config.baseUrl}/api/tags`, {
        signal: AbortSignal.timeout(5000),
      });
      if (!res.ok) return { ok: false, models: [] };
      const data = await res.json();
      const models = (data.models ?? []).map((m: { name: string }) => m.name);
      return { ok: true, models };
    } catch {
      return { ok: false, models: [] };
    }
  }

  // Pull a model if not present
  async pullModel(model: string, onProgress?: (status: string) => void): Promise<void> {
    const response = await fetch(`${this.config.baseUrl}/api/pull`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ name: model, stream: true }),
    });

    const reader = response.body!.getReader();
    const decoder = new TextDecoder();

    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      const lines = decoder.decode(value).split('\n').filter(Boolean);
      for (const line of lines) {
        try {
          const chunk = JSON.parse(line);
          onProgress?.(chunk.status ?? '');
        } catch { /* skip */ }
      }
    }
  }
}

export class OllamaError extends Error {
  constructor(message: string) {
    super(message);
    this.name = 'OllamaError';
  }
}
```

```typescript
import { getAI, ChatMessage } from '../ai/client';

export interface AgentContext {
  userId: string;
  sessionId: string;
  memory?: string;     // Injected long-term memory summary
  rag?: string;        // Injected RAG context
}

export abstract class BaseAgent {
  abstract name: string;
  abstract systemPrompt: string;

  protected buildMessages(userMessage: string, ctx: AgentContext): ChatMessage[] {
    const system = [
      this.systemPrompt,
      ctx.memory ? `\n\n## Memory Context\n${ctx.memory}` : '',
      ctx.rag ? `\n\n## Retrieved Knowledge\n${ctx.rag}` : '',
    ].join('');

    return [
      { role: 'system', content: system },
      { role: 'user', content: userMessage },
    ];
  }

  async run(userMessage: string, ctx: AgentContext): Promise<string> {
    const ai = getAI();
    const messages = this.buildMessages(userMessage, ctx);
    return ai.chat(messages);
  }

  async *stream(userMessage: string, ctx: AgentContext) {
    const ai = getAI();
    const messages = this.buildMessages(userMessage, ctx);
    yield* ai.chatStream(messages);
  }
}
```

**Step 4.1 — Streaming Endpoint**

Update or create `apps/api/routes/chat.ts` (or equivalent):

```typescript
// apps/api/routes/chat.ts

import { getAI } from '@onefoudner/ai/client';
import { agentRegistry } from '@onefoudner/agents';
import { memoryService } from '@onefoudner/memory';

export async function POST(req: Request) {
  const { message, agentId, userId, sessionId } = await req.json();

  const agent = agentRegistry.get(agentId);
  if (!agent) return new Response('Agent not found', { status: 404 });

  const memory = await memoryService.getContext(userId, sessionId);
  const ctx = { userId, sessionId, memory };

  // Check if client wants streaming
  const acceptsStream = req.headers.get('accept') === 'text/event-stream';

  if (acceptsStream) {
    const encoder = new TextEncoder();
    const stream = new ReadableStream({
      async start(controller) {
        for await (const chunk of agent.stream(message, ctx)) {
          controller.enqueue(
            encoder.encode(`data: ${JSON.stringify(chunk)}\n\n`)
          );
          if (chunk.done) break;
        }
        controller.close();
      },
    });

    return new Response(stream, {
      headers: {
        'Content-Type': 'text/event-stream',
        'Cache-Control': 'no-cache',
        'Connection': 'keep-alive',
      },
    });
  }

  // Non-streaming fallback
  const response = await agent.run(message, ctx);
  return Response.json({ response });
}
```

```typescript
// packages/memory/service.ts

import { db } from '../db/client';

export class MemoryService {
  async getContext(userId: string, sessionId: string): Promise<string> {
    const memories = await db.memory.findMany({
      where: { userId },
      orderBy: { createdAt: 'desc' },
      take: 20,
    });

    if (memories.length === 0) return '';

    return memories
      .map(m => `[${m.type}] ${m.content}`)
      .join('\n');
  }

  async save(params: {
    userId: string;
    sessionId: string;
    type: 'conversation' | 'founder' | 'preference' | 'knowledge';
    content: string;
  }): Promise<void> {
    await db.memory.create({ data: params });
  }

  async summarizeAndCompress(userId: string): Promise<void> {
    // Get old conversation memories
    const old = await db.memory.findMany({
      where: { userId, type: 'conversation' },
      orderBy: { createdAt: 'asc' },
      take: 50,
    });

    if (old.length < 20) return;

    const ai = (await import('../ai/client')).getAI();
    const summary = await ai.chat([
      {
        role: 'system',
        content: 'Summarize the following conversation memories into a compact 3-5 sentence founder profile. Keep key facts, preferences, and business context.',
      },
      {
        role: 'user',
        content: old.map(m => m.content).join('\n'),
      },
    ]);

    // Delete old, save summary
    await db.memory.deleteMany({ where: { id: { in: old.map(m => m.id) } } });
    await this.save({ userId, sessionId: 'compressed', type: 'founder', content: summary });
  }
}

export const memoryService = new MemoryService();
```

```typescript
export async function embed(text: string): Promise<number[]> {
  const res = await fetch(`${OLLAMA_URL}/api/embeddings`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ model: EMBED_MODEL, prompt: text }),

  if (!res.ok) throw new Error(`Embedding failed: ${res.status}`);
  const data = await res.json();
  return data.embedding as number[];
}

export function cosineSimilarity(a: number[], b: number[]): number {
  const dot = a.reduce((sum, ai, i) => sum + ai * b[i], 0);
  const magA = Math.sqrt(a.reduce((sum, ai) => sum + ai * ai, 0));
  const magB = Math.sqrt(b.reduce((sum, bi) => sum + bi * bi, 0));
  return dot / (magA * magB);
}
```

```typescript
  if (action === 'pull') {
    const ai = getAI();
    // Stream pull progress back
    const encoder = new TextEncoder();
    const stream = new ReadableStream({
      async start(controller) {

      },
```

### PHASE 9 — Performance Optimizations

**Step 9.1 — Response Caching**

Create `packages/ai/cache.ts`:

```typescript
// packages/ai/cache.ts
// Simple in-memory + SQLite cache for repeated prompts

import crypto from 'crypto';

const memCache = new Map<string, { response: string; ts: number }>();
const TTL = 1000 * 60 * 30; // 30 minutes

export function cacheKey(model: string, messages: object[]): string {
  return crypto
    .createHash('sha256')
    .update(model + JSON.stringify(messages))
    .digest('hex');
}

export function getCached(key: string): string | null {
  const entry = memCache.get(key);
  if (!entry) return null;
  if (Date.now() - entry.ts > TTL) {
    memCache.delete(key);
    return null;
  }
  return entry.response;
}

export function setCache(key: string, response: string): void {
  memCache.set(key, { response, ts: Date.now() });
  // Keep cache size bounded
  if (memCache.size > 200) {
    const oldest = [...memCache.entries()].sort((a, b) => a[1].ts - b[1].ts)[0];
    memCache.delete(oldest[0]);
  }
}
```

**Step 9.2 — Context Compression**

Add to `BaseAgent`:

```typescript
// In packages/agents/base.ts — add this helper

protected compressContext(messages: ChatMessage[], maxTokens = 2000): ChatMessage[] {
  // Rough token estimate: 1 token ≈ 4 chars
  let total = messages.reduce((sum, m) => sum + m.content.length / 4, 0);

  if (total <= maxTokens) return messages;

  // Keep system prompt + last N user/assistant turns
  const [system, ...rest] = messages;
  const compressed: ChatMessage[] = [system];

  for (let i = rest.length - 1; i >= 0 && total > maxTokens; i--) {
    total -= rest[i].content.length / 4;
  }

  return [...compressed, ...rest.slice(-6)]; // Always keep last 3 turns
}
```

## 🔗 Replit-Specific Notes

| Issue | Solution |
|-------|----------|
| Ollama can't run inside Replit | Use local machine + ngrok/Tailscale |
| Port 11434 not accessible | Expose via `ngrok http 11434` |
| Secrets not loading | Add via Replit Secrets panel, not `.env` file |
| CORS errors | Add Ollama origin to CORS config: `OLLAMA_ORIGINS=*` on the Ollama server |
| Replit sleeps the app | Ollama runs on user's machine — unaffected |

**Enabling CORS on Ollama (user's machine):**

```bash
# macOS/Linux
OLLAMA_ORIGINS="*" ollama serve

# Windows (PowerShell)
$env:OLLAMA_ORIGINS="*"; ollama serve
```
